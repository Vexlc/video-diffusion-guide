def crossing(upper_probability):
    """两道半程相交的题，分别给出向右上、向右下的标签。"""
    upper, lower = (4, 2), (4, -2)
    p = upper_probability
    mean = tuple(p*a + (1-p)*b for a,b in zip(upper, lower))

    def loss(guess):
        upper_error = sum((g-u)**2 for g,u in zip(guess,upper))
        lower_error = sum((g-u)**2 for g,u in zip(guess,lower))
        return p*upper_error + (1-p)*lower_error

    return mean, loss(upper), loss(mean)

def predict_clean(state, candidates, signal, sigma):
    """比较四条候选轨迹，越符合带噪观测的轨迹获得越高权重。"""
    logits = [-sum((value-signal*target)**2 for value, target in zip(state, path))
              / (2*sigma**2) for path in candidates]
    peak = max(logits)
    weights = [math.exp(logit-peak) for logit in logits]
    return [sum(weight * position for weight, position in zip(weights, column)) / sum(weights)
            for column in zip(*candidates)]

def gate_updates():
    """冻结基座；固定支路输入 r，观察出口更新前后，损失能否传到 r。"""
    base, r, target = 0.2, 2.0, 1.0
    weight, bias, learning_rate = 0.0, 0.0, 0.1
    for step in range(2):
        prediction = base + weight*r + bias
        error = prediction-target                  # 损失为 (预测−目标)²/2
        grad_weight, grad_bias = error*r, error
        grad_r = error*weight
        yield step, weight, bias, prediction, grad_weight, grad_r
        weight -= learning_rate*grad_weight
        bias -= learning_rate*grad_bias

def reverse_step(state, clean, tick, total, sampler, rng):
    signal, sigma = schedule(tick/total)
    prev_signal, prev_sigma = schedule((tick-1)/total)
    if sampler == "DDIM":
        noise = [(value-signal*estimate)/sigma for value, estimate in zip(state, clean)]
        return [prev_signal*estimate + prev_sigma*eps for estimate, eps in zip(clean, noise)]
    if sampler != "DDPM":
        raise ValueError("unknown sampler")
    alpha = (signal/prev_signal)**2
    beta = 1-alpha
    step_sigma = math.sqrt(beta * prev_sigma**2 / sigma**2)
    means = [(prev_signal*beta*estimate + math.sqrt(alpha)*prev_sigma**2*value)/sigma**2
             for value, estimate in zip(state, clean)]
    return [center + step_sigma*rng.gauss(0, 1) for center in means] if tick > 1 else means

def mix_neighbors(feature, frames, enabled):
    if not enabled:
        return feature
    chunks = feature.reshape(-1, frames, *feature.shape[1:])
    piece = chunks.shape[2]//4
    previous = torch.cat([chunks[:, :1, :piece], chunks[:, :-1, :piece]], dim=1)
    following = torch.cat([chunks[:, 1:, piece:2*piece], chunks[:, -1:, piece:2*piece]], dim=1)

    """每个区间用平方损失拟合一个位移均值，将节点间线性连接。"""

def gate(samples, x, t, half_span):
    early,late=max(0,t-half_span),min(1,t+half_span)
    # q = Φ(z) 沿精确一维 ODE 保持不变。q < F_t(x) 当且仅当轨迹位于门左侧。
    def cutoff(time):
        prob=cdf(x,time)
        if prob==0: return -math.inf
        if prob==1: return math.inf
        return NORMAL.inv_cdf(prob)

    cutoff_early,cutoff_late=cutoff(early),cutoff(late)
    before=after=right=left=ode_before=ode_after=0

        start=(1-early)*z+early*y < x
        end=(1-late)*z+late*y < x

        after+=end

        left+=not start and end

def line_sample(model, initial, steps, *, uniform=False, raw=False):
    """每个分支从自己的状态调用网络。与原版相配的节点按 tau=a/(a+sigma) 排列。"""
    nodes = []
    for i in range(steps+1):

        a, b = math.cos(angle), math.sin(angle)
        nodes.append(i/steps if uniform else a/(a+b))
    nodes[0], nodes[-1] = 0., 1.
    state = initial.clone()
    states = [state]
    for tau, following in zip(nodes, nodes[1:]):
        a, b = coefficients(tau)
        tick = 2/math.pi*math.atan2(b,a)
        original_units = (a+b)*state
        clock = torch.full((len(state),), tick, dtype=state.dtype, device=state.device)
        v = model(original_units, clock)
        _, _, speed = line_coordinates(original_units, v, a, b)
        state = state + (following-tau)*(v if raw else speed)
        states.append(state)
    return nodes, states

def noise_problem(noise_error):
    original = square(frame=2, row=3)
    rng = Random(17)
    noise = [[rng.gauss(0,1) for x in range(16)] for y in range(12)]
    a = 0.8
    sigma = sqrt(1-a*a)
    mixed = [[a*y + sigma*z for y,z in zip(pixels,snow)]
             for pixels,snow in zip(original,noise)]
    # 已知这次出题用的噪声，给每一格的猜测故意加同样的误差。
    restored = [[(x-sigma*(z+noise_error))/a for x,z in zip(pixels,snow)]
                for pixels,snow in zip(mixed,noise)]
    return original, mixed, restored
